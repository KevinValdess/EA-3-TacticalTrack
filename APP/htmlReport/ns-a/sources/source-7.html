 @OptIn(ExperimentalMaterial3Api::class)
 @Composable
 fun SupervisorScreen(
     viewModel: VisitViewModel
 ) {
     val remote by viewModel.remote.collectAsStateWithLifecycle()
     val visits by viewModel.visits.collectAsStateWithLifecycle()
 
     Scaffold(
         topBar = {
             TopAppBar(title = { Text("Validación de supervisor") }, actions = {
                 androidx.compose.material3.TextButton(onClick = viewModel::refreshRemote, enabled = !remote.busy) { Text("Actualizar") }
             })
         }
     ) { innerPadding ->
         if (remote.error != null) {
             Text(remote.error.orEmpty(), Modifier.padding(innerPadding).padding(16.dp), color = MaterialTheme.colorScheme.error)
         } else if (visits.isEmpty()) {
             Text(
                 "No existen registros para revisar.",
                 modifier = Modifier
                     .fillMaxSize()
                     .padding(innerPadding)
                     .padding(24.dp)
             )
         } else {
             LazyColumn(
                 modifier = Modifier
                     .fillMaxSize()
                     .padding(innerPadding)
                     .padding(horizontal = 12.dp),
                 verticalArrangement = Arrangement.spacedBy(10.dp)
             ) {
                 items(visits, key = { it.id }) { visit ->
                     Card(Modifier.fillMaxWidth()) {
                         Column(
                             Modifier.padding(16.dp),
                             verticalArrangement = Arrangement.spacedBy(8.dp)
                         ) {
                             Text(
                                 "${visit.assetCode} · ${visit.pointName}",
                                 style = MaterialTheme.typography.titleMedium
                             )
                             Text("Operario: ${visit.operatorName}")
                             Text(
                                 if (visit.presenceVerified) {
                                     "Ubicación dentro del radio de 150 m"
                                 } else {
                                     "Ubicación fuera del radio de 150 m"
                                 }
                             )
                             Text("Estado: ${visit.supervisorStatus}")
 
                             Row(
                                 modifier = Modifier.fillMaxWidth(),
                                 horizontalArrangement = Arrangement.spacedBy(8.dp)
                             ) {
                                 Button(
                                     onClick = { viewModel.approveVisit(visit.id) },
                                     enabled = !remote.busy,
                                     modifier = Modifier.weight(1f)
                                 ) {
                                     Icon(Icons.Default.Check, contentDescription = null)
                                     Text(" Aprobar")
                                 }
                                 OutlinedButton(
                                     onClick = { viewModel.rejectVisit(visit.id) },
                                     enabled = !remote.busy,
                                     modifier = Modifier.weight(1f)
                                 ) {
                                     Icon(Icons.Default.Close, contentDescription = null)
                                     Text(" Rechazar")
                                 }
                             }
                         }
                     }
                 }
             }
         }
     }
 }
