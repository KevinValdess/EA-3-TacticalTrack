 @OptIn(ExperimentalMaterial3Api::class)
 @Composable
 fun VisitDetailScreen(
     visitId: Long,
     viewModel: VisitViewModel,
     onBack: () -> Unit
 ) {
     val remote by viewModel.remote.collectAsStateWithLifecycle()
     var editing by remember(visitId) { mutableStateOf(false) }
     var deleting by remember(visitId) { mutableStateOf(false) }
     val detailState = key(visitId, viewModel) {
         val flow = remember(visitId, viewModel) { viewModel.observeVisitDetail(visitId) }
         val state by flow.collectAsStateWithLifecycle(initialValue = VisitDetailUiState.Loading)
         state
     }
 
     Scaffold(
         topBar = {
             TopAppBar(
                 title = { Text("Detalle de visita") },
                 navigationIcon = {
                     IconButton(onClick = onBack) {
                         Icon(Icons.Default.ArrowBack, contentDescription = "Volver")
                     }
                 }
             )
         }
     ) { innerPadding ->
         val current = (detailState as? VisitDetailUiState.Found)?.visit
         if (current == null) {
             Text(
                 if (detailState == VisitDetailUiState.NotFound) "Visita no encontrada" else "Cargando visita...",
                 modifier = Modifier
                     .fillMaxSize()
                     .padding(innerPadding)
                     .padding(24.dp)
             )
         } else {
             val date = SimpleDateFormat(
                 "dd/MM/yyyy HH:mm:ss",
                 Locale.getDefault()
             ).format(Date(current.timestamp))
 
             Column(
                 modifier = Modifier
                     .fillMaxSize()
                     .padding(innerPadding)
                     .verticalScroll(rememberScrollState())
                     .padding(16.dp),
                 verticalArrangement = Arrangement.spacedBy(12.dp)
             ) {
                 remote.error?.let { Text(it, color = MaterialTheme.colorScheme.error) }
                 Button(onClick = { editing = true }, enabled = !remote.busy) { Text("Editar visita") }
                 OutlinedButton(onClick = { deleting = true }, enabled = !remote.busy) { Text("Eliminar visita") }
                 if (editing) {
                     EditVisitDialog(current.equipmentStatus, current.observations, remote.busy,
                         serverError = remote.error,
                         onDismiss = { editing = false },
                         onSave = { status, observations ->
                             viewModel.editVisit(visitId, status, observations) { editing = false }
                         })
                 }
                 if (deleting) {
                     AlertDialog(onDismissRequest = { if (!remote.busy) deleting = false },
                         title = { Text("¿Eliminar visita?") },
                         text = { Column {
                             Text("Se eliminará del servidor. Esta acción no se puede deshacer.")
                             remote.error?.let { Text(it, color = MaterialTheme.colorScheme.error) }
                         } },
                         confirmButton = { TextButton(enabled = !remote.busy, onClick = {
                             viewModel.deleteVisit(visitId) { deleting = false; onBack() }
                         }) { Text("Eliminar") } },
                         dismissButton = { TextButton(enabled = !remote.busy, onClick = { deleting = false }) { Text("Cancelar") } })
                 }
                 cl.duoc.tacticaltrack.ui.components.WeatherCard(current.latitude, current.longitude)
                 Text(
                     "${current.assetCode} · ${current.assetType}",
                     style = MaterialTheme.typography.headlineSmall
                 )
                 VisitStatusChips(
                     current.presenceVerified,
                     current.supervisorStatus,
                     current.synced
                 )
 
                 AsyncImage(
                     model = current.photoUri,
                     contentDescription = "Fotografía de evidencia",
                     contentScale = ContentScale.Crop,
                     modifier = Modifier
                         .fillMaxWidth()
                         .height(240.dp)
                         .clip(RoundedCornerShape(16.dp))
                 )
 
                 Card(Modifier.fillMaxWidth()) {
                     Column(
                         Modifier.padding(16.dp),
                         verticalArrangement = Arrangement.spacedBy(5.dp)
                     ) {
                         Text("Punto: ${current.pointName}")
                         Text("Operario: ${current.operatorName} (${current.operatorRole})")
                         Text("Fecha y hora: $date")
                         Text("Tipo de visita: ${current.visitType}")
                         Text("Estado del equipo: ${current.equipmentStatus}")
                         Text("Observaciones: ${current.observations.ifBlank { "Sin observaciones" }}")
                         Text("Latitud: ${current.latitude}")
                         Text("Longitud: ${current.longitude}")
                         Text("Distancia al punto: ${"%.1f".format(current.distanceToPointMeters)} m")
                     }
                 }
             }
         }
     }
 }
